10 REM *** PC-1500 BMP280: TEMPERATURE + PRESSURE, FORCED MODE ***
20 REM NEEDS I2C V1.4. SCL=PC7 SDA=PB2. BMP280 AT &77 (SDO=VDDIO)
30 CLEAR : WAIT 0 : D = &77
40 A0 = (PEEK &7863) * 256 : IC = A0 + &400 : IA = A0 + &4FC
50 TX = A0 + &300 : RX = A0 + &200
60 IF PEEK (IC) <> &FD OR PEEK (IC + 1) <> &BE PRINT "I2C NOT INSTALLED" : END
100 REM --- INITIALISING: CHIP ID, SOFT RESET, CALIBRATION ---
110 WAIT 64 : PRINT "Initialising..." : WAIT 0
120 POKE TX, &D0 : POKE IA, D, 1, 1 : CALL IC, E : IF E <> 0 GOTO 990
130 I = PEEK (RX) : IF (I < &56 OR I > &58) AND I <> &60 GOTO 980
140 POKE TX, &E0, &B6 : POKE IA, D, 2, 0 : CALL IC, E : IF E <> 0 GOTO 990
150 FOR I = 1 TO 50 : NEXT I : K = 0
160 GOSUB 700 : K = K + 1 : IF (S AND 1) <> 0 AND K < 50 GOTO 160
170 POKE TX, &88 : POKE IA, D, 1, 24 : CALL IC, E : IF E <> 0 GOTO 990
200 N = 0 : GOSUB 900 : T1 = X : N = 2 : GOSUB 910 : T2 = X
210 N = 4 : GOSUB 910 : T3 = X : N = 6 : GOSUB 900 : P1 = X
220 N = 8 : GOSUB 910 : P2 = X : N = 10 : GOSUB 910 : P3 = X
230 N = 12 : GOSUB 910 : P4 = X : N = 14 : GOSUB 910 : P5 = X
240 N = 16 : GOSUB 910 : P6 = X : N = 18 : GOSUB 910 : P7 = X
250 N = 20 : GOSUB 910 : P8 = X : N = 22 : GOSUB 910 : P9 = X
300 REM --- SETUP: OVERSAMPLING AND IIR FILTER (ENTER = 0) ---
310 WAIT 128 : PRINT "0,1,2,4,8,16 ENTER=0" : WAIT 0
320 X = 0 : INPUT "temp oversampling ? "; X : GOSUB 800 : IF C < 0 GOTO 320
330 CT = C
340 X = 0 : INPUT "Pres oversampling ? "; X : GOSUB 800 : IF C < 0 GOTO 340
350 CP = C
360 X = 0 : INPUT "IR Filtering ? "; X : GOSUB 800 : IF C < 0 GOTO 360
365 IF C > 0 LET C = C - 1
370 CF = C
380 IF CT = 0 LET CT = 1 : WAIT 128 : PRINT "Temp x1 (needed for P)" : WAIT 0
390 POKE TX, &F5, CF * 4 : POKE IA, D, 2, 0 : CALL IC, E : IF E <> 0 GOTO 990
400 REM --- ONE MEASUREMENT: FORCED MODE (MODE = 01) ---
410 POKE TX, &F4, CT * 32 + CP * 4 + 1 : POKE IA, D, 2, 0 : CALL IC, E
420 IF E <> 0 GOTO 990
430 K = 0
440 GOSUB 700 : K = K + 1 : IF (S AND 8) <> 0 AND K < 200 GOTO 440
445 IF K >= 200 PRINT "BMP280 TIMEOUT" : END
450 POKE TX, &F7 : POKE IA, D, 1, 6 : CALL IC, E : IF E <> 0 GOTO 990
460 UP = PEEK (RX) * 4096 + PEEK (RX + 1) * 16 + INT (PEEK (RX + 2) / 16)
470 UT = PEEK (RX + 3) * 4096 + PEEK (RX + 4) * 16 + INT (PEEK (RX + 5) / 16)
480 REM --- COMPENSATION (DATASHEET, FLOATING POINT) ---
485 V1 = (UT / 16384 - T1 / 1024) * T2
490 V2 = UT / 131072 - T1 / 8192 : V2 = V2 * V2 * T3
495 TF = V1 + V2 : TC = TF / 5120
510 IF CP = 0 GOTO 600
515 V1 = TF / 2 - 64000 : V2 = V1 * V1 * P6 / 32768
520 V2 = V2 + V1 * P5 * 2 : V2 = V2 / 4 + P4 * 65536
525 V1 = (P3 * V1 * V1 / 524288 + P2 * V1) / 524288
530 V1 = (1 + V1 / 32768) * P1 : PA = 0 : IF V1 = 0 GOTO 600
535 PA = 1048576 - UP : PA = (PA - V2 / 4096) * 6250 / V1
540 V1 = P9 * PA * PA / 2147483648 : V2 = PA * P8 / 32768
545 PA = PA + (V1 + V2 + P7) / 16
600 REM --- DISPLAY: T, THEN P; ANY KEY STOPS ---
610 WAIT 128 : PRINT "Temperature ="; USING "###.##"; TC; " C"
620 IF CP = 0 GOTO 640
630 PRINT "Pressure ="; USING "#####.##"; PA / 100; " hPa"
640 IF INKEY$ = "" GOTO 410
650 WAIT 0 : PRINT "STOPPED" : END
700 REM --- STATUS REGISTER &F3 INTO S ---
710 POKE TX, &F3 : POKE IA, D, 1, 1 : CALL IC, E : IF E <> 0 GOTO 990
720 S = PEEK (RX) : RETURN
800 REM --- X (0,1,2,4,8,16) INTO CODE C (0..5), C = -1 IF WRONG ---
810 C = -1 : V = 1 : FOR J = 1 TO 5 : IF X = V LET C = J
820 V = V * 2 : NEXT J : IF X = 0 LET C = 0
830 RETURN
900 REM --- 16-BIT VALUE FROM RX+N (LSB FIRST) INTO X ---
905 X = PEEK (RX + N) + 256 * PEEK (RX + N + 1) : RETURN
910 GOSUB 900 : IF X > 32767 LET X = X - 65536
915 RETURN
980 WAIT 0 : PRINT "BAD CHIP ID "; I : END
990 WAIT 0 : PRINT "I2C ERROR E="; E : BEEP 3 : END
