10 REM *** PC-1500 UART 19200 CALIBRATION v1.0 (WITH ARDUINO UNO) ***
20 REM NEEDS SERINOUT v7.0 (INSTALLER v7.0_19200, NORMAL POLARITY)
30 REM AND THE ARDUINO SKETCH UART_CALIBRATION. NEW &4400 IN PRO MODE
40 A0 = (PEEK &7863) * 256 : TX = A0 + &300 : RX = A0 + &200
50 SO = A0 + &C5 : SI = A0 + &130 : TP = INT (TX / 256) : RP = TP - 1
60 IF PEEK (SI + 10) <> &AE PRINT "NO SERINOUT v7.0! RUN 10" : END
70 TB = PEEK (A0 + &1FC) : RB = PEEK (A0 + &1FD) : RQ = PEEK (A0 + &1FE)
80 WAIT 0 : PRINT "TB="; TB; " RB="; RB; " RQ="; RQ
82 T = 0 : INPUT "NEW TIMING (1=YES)? "; T
84 IF T = 1 GOSUB 300
90 INPUT "ARDUINO READY? ENTER "; Z$
100 POKE TX, 67, 65, 76, TB, RB, RQ, 1 : N = 7 : CALL SO, N : REM "CAL"
110 PRINT "TX TEST: 64 X 'U'..."
120 FOR I = 0 TO 63 : POKE TX + I, 85 : NEXT I
130 N = 64 : CALL SO, N
140 POKE SO + 20, RP : REM SEROUT SENDS FROM THE RX BUFFER (ECHO)
150 K = 0 : PRINT "ECHO LOOP..."
160 POKE RX, 82 : CALL SO : REM "R" = READY FOR A BLOCK
170 N = 0 : CALL SI, N : IF N = 0 GOTO 210
180 CALL SO, N : K = K + 1
190 IF K = 10 * INT (K / 10) PRINT "BLOCKS: "; K
200 GOTO 160
205 REM SEROUT SENDS FROM TX AGAIN (AFTER BREAK: RUN 210)
210 A0 = (PEEK &7863) * 256 : POKE A0 + &D9, A0 / 256 + 3
220 BEEP 1 : WAIT : PRINT "END: "; K; " BLOCKS" : END
300 REM --- NEW TIMING: WRITE THE PADS INTO SEROUT/SERIN ---
310 INPUT "TX BIT TB (63-67)? "; TB
315 IF TB < 63 OR TB > 67 BEEP 1 : GOTO 310
320 INPUT "RX BIT RB (63-71)? "; RB
325 IF RB < 63 OR RB > 71 BEEP 1 : GOTO 320
330 INPUT "RX POINT RQ (8,12-20)? "; RQ
335 IF RQ <> 8 AND (RQ < 12 OR RQ > 20) BEEP 1 : GOTO 330
340 V = TB - 57 : P1 = &38 : P2 = &38 : REM 10 CYCLES: NOP NOP
342 IF V = 6 THEN LET P1 = &6A : LET P2 = 0
344 IF V = 7 THEN LET P1 = &6E : LET P2 = 0
346 IF V = 8 THEN LET P1 = &8E : LET P2 = 0
348 IF V = 9 THEN LET P1 = &F9
350 POKE SO + 38, P1, P2 : POKE SO + 59, P1, P2
360 V = RQ : GOSUB 900 : POKE SI + 85, W1, W2, W3, W4
370 V = RB - 51 : GOSUB 900 : POKE SI + 89, W1, W2, W3, W4
380 POKE A0 + &1FC, TB, RB, RQ : PRINT "TB="; TB; " RB="; RB; " RQ="; RQ
390 RETURN
900 REM --- 4-BYTE TIMING PAD OF V CYCLES (8, 12..20) -> W1..W4 ---
905 W1 = &38 : W2 = &38 : W3 = &38 : W4 = &38 : REM 20: 4 X NOP
910 IF V = 8 THEN LET W1 = &8E : LET W2 = 2 : REM BCH+ 2
912 IF V = 12 THEN LET W1 = &F9 : LET W2 = &8E : LET W3 = 1
914 IF V = 13 THEN LET W2 = &8E : LET W3 = 1
916 IF V = 14 THEN LET W1 = &F9 : LET W2 = &6A : LET W3 = 0 : LET W4 = &F9
918 IF V = 15 THEN LET W2 = &6A : LET W3 = 0 : LET W4 = &F9
920 IF V = 16 THEN LET W2 = &6A : LET W3 = 0
922 IF V = 17 THEN LET W2 = &6E : LET W3 = 0
924 IF V = 18 THEN LET W2 = &8E : LET W3 = 0
926 IF V = 19 THEN LET W2 = &F9
928 RETURN
