10 REM *** PC-1500 QUOTE OF THE DAY v1.0 (SERINOUT v7.2, ARDUINO UNO) ***
20 REM ASKS THE ARDUINO SKETCH UART_QUOTE "WHAT'S FOR TODAY?" AND SHOWS
30 REM ITS ANSWER. FIRST THE INSTALLER v7.2 (SPEED 5), THEN NEW AND THIS
40 CLEAR : DIM W$(0) * 26, A$(0) * 26 : WAIT 0
50 A0 = (PEEK &7863) * 256 : TX = A0 + &300 : RX = A0 + &200
60 SO = A0 + &C5 : SI = A0 + &130
70 BP = (PEEK &7865) * 256 + PEEK &7866
80 IF BP < A0 + &400 PRINT "NEW &4400 IN PRO MODE!" : BEEP 3 : END
90 REM --- SPEED AND POLARITY OF THE INSTALLED SERINOUT ---
100 BD = 0 : K = PEEK (SI + 77) : F = PEEK (SI + 36)
105 IF K = 91 THEN LET BD = 1200
110 IF K = 42 THEN LET BD = 2400
115 IF K = 17 THEN LET BD = 4800
120 IF K = 5 THEN LET BD = 9600
125 IF PEEK (SO + 59) <> K THEN LET BD = 0
130 IF PEEK (SO + 18) <> &AE OR PEEK (SI + 85) <> &8E GOTO 140
135 BD = 19200 : F = PEEK (SI + 33)
140 IF BD = 0 PRINT "NO SERINOUT! RUN INSTALLER" : END
145 IF F = &8B PRINT "INSTALL WITH INVERSION=NO!" : END
150 POKE# &F008, (PEEK# &F008) OR &80 : REM TX IDLE = HIGH
160 WAIT 96 : PRINT "QUOTE OF THE DAY "; BD; " BPS"
200 REM --- QUESTION ---
210 WAIT 64 : PRINT "1=TODAY 2=TOMORROW"
220 WAIT 0 : D = 1 : INPUT "3=YESTERDAY 0=END? "; D
230 IF D = 0 GOTO 900
240 IF D < 1 OR D > 3 OR D <> INT (D) BEEP 1 : GOTO 210
250 Q$ = "TODAY" : IF D = 2 THEN LET Q$ = "TOMORROW"
260 IF D = 3 THEN LET Q$ = "YESTERDAY"
270 W$(0) = "WHAT'S FOR " + Q$ + "?" : N = LEN (W$(0))
280 REM SEND "What's for ...?" (LOWER CASE AFTER THE FIRST LETTER) AND CR
290 FOR I = 1 TO N : C = ASC (MID$ (W$(0), I, 1))
300 IF I > 1 AND C > 64 AND C < 91 THEN LET C = C + 32
310 POKE TX + I - 1, C : NEXT I
320 POKE TX + N, 13 : N = N + 1
330 PRINT "ASKING..." : CALL SO, N
340 M = 27 : CALL SI, M : REM M <- NUMBER OF CHARACTERS RECEIVED
350 IF M = 0 BEEP 2 : WAIT : PRINT "NO ANSWER (ARDUINO?)" : GOTO 210
360 REM --- ANSWER: PRINTABLE CHARACTERS, AT MOST ONE LINE ---
370 A$(0) = "" : FOR I = 0 TO M - 1 : C = PEEK (RX + I)
380 IF C < 32 OR C > 126 OR LEN (A$(0)) > 25 GOTO 390
385 A$(0) = A$(0) + CHR$ (C)
390 NEXT I
400 BEEP 1 : WAIT : PRINT A$(0)
410 GOTO 210
900 WAIT 0 : PRINT "BYE" : END
