10 REM ****** PC-1500 UART LINK v1.0 (NEEDS SERIN/SEROUT v6.x) ******
20 REM WIRES: PC7 (PIN 10) -> PB0 (PIN 9) OR PB2 (PIN 27), GND (52-55)
30 REM BOTH PC-1500: SAME BAUD RATE AND POLARITY (INSTALLER v6.2)
40 CLEAR : DIM M$(0)*80, L$(0)*26 : RANDOM : WAIT 0
50 A0 = (PEEK &7863) * 256 : SO = A0 + &C5 : SI = A0 + &130
60 RC = A0 + &1FF : RX = A0 + &200 : TX = A0 + &300
70 BP = (PEEK &7865) * 256 + PEEK &7866
80 IF BP < A0 + &400 PRINT "NEW &4400 IN PRO MODE!" : BEEP 3 : END
90 REM --- READ SPEED, RX PORT AND POLARITY FROM THE INSTALLED CODE ---
100 K = PEEK (SI + 77) : Q = PEEK (SI + 35) : BD = 0 : P$ = "" : V = 0
105 IF K = 91 THEN LET BD = 1200
110 IF K = 42 THEN LET BD = 2400
115 IF K = 17 THEN LET BD = 4800
120 IF K = 5 THEN LET BD = 9600
125 IF Q = 1 THEN LET P$ = "PB0"
130 IF Q = 4 THEN LET P$ = "PB2"
135 IF PEEK (SO + 59) <> K THEN LET BD = 0
140 IF BD = 0 OR P$ = "" GOTO 960
145 IF PEEK (SI + 36) = &8B THEN LET V = 1 : LET P$ = P$ + " INV"
150 IF V = 1 POKE# &F008, (PEEK# &F008) AND &7F : REM INVERTED: IDLE = LOW
155 IF V = 0 POKE# &F008, (PEEK# &F008) OR &80 : REM NORMAL: IDLE = HIGH
160 PL = INT (BD / 64) : REM TX PREAMBLE: PL NUL BYTES = ABOUT 150 MS
165 WAIT 96 : PRINT "UART LINK "; BD; " "; P$
200 REM --- MENU ---
205 POKE SI + 32, 18 : REM SERIN FIRST-CHAR TIMEOUT BACK TO ~30 S
210 WAIT 0 : S = 9 : INPUT "1=DEMO 2=TX 3=RX 0=END "; S
215 IF S = 0 GOTO 950
220 IF S = 1 GOTO 300
225 IF S = 2 GOTO 400
230 IF S = 3 GOTO 500
235 BEEP 1 : GOTO 210
300 REM --- 1 = DEMO: SEND A RANDOM 3-DIGIT NUMBER EVERY 5 S ---
305 GOSUB 800 : C = 0
310 GOSUB 850 : TS = T : R = RND (900) + 99 : REM R = 100..999
315 H = INT (R / 100) : Y = INT (R / 10) - 10 * H : Z = R - 10 * INT (R / 10)
320 POKE TX + PL, 48 + H, 48 + Y, 48 + Z, 13, 10 : REM 3 DIGITS, CR, LF
325 N = PL + 5 : CALL SO, N : C = C + 1
330 WAIT 0 : PRINT "TX #"; C; ": "; R; "  KEY=MENU"
335 GOSUB 850 : E = T - TS : IF E < 0 THEN LET E = E + 3600
340 IF E >= 5 GOTO 310
345 G$ = INKEY$ : IF G$ = "" GOTO 335
350 GOTO 900
400 REM --- 2 = SEND MY OWN TEXT (UP TO 80 CHARS), ENTER ALONE = MENU ---
405 GOSUB 800 : C = 0
410 M$(0) = "" : INPUT "TX> "; M$(0)
415 IF M$(0) = "" GOTO 200
420 L = LEN (M$(0))
425 FOR I = 1 TO L : POKE TX + PL + I - 1, ASC (MID$ (M$(0), I, 1)) : NEXT I
430 POKE TX + PL + L, 13, 10 : N = PL + L + 2 : CALL SO, N : C = C + 1
435 WAIT 64 : PRINT "SENT #"; C; ": "; L; " CHARS" : GOTO 410
500 REM --- 3 = RECEIVE AND SHOW; HOLD ANY KEY ~2 S = MENU ---
505 POKE SI + 32, 1 : REM SERIN RETURNS AFTER ~1.7 S WITHOUT DATA
510 C = 0 : WAIT 0 : PRINT "RX "; BD; " "; P$; " HOLD KEY"
515 CALL SI : D = PEEK RC : IF D > 0 GOTO 530
520 G$ = INKEY$ : IF G$ = "" GOTO 515
525 GOTO 900
530 U = 0 : W = D : REM BINARY SEARCH: FIRST NON-NUL BYTE (SKIP PREAMBLE)
535 IF U >= W GOTO 550
540 I = INT ((U + W) / 2) : IF PEEK (RX + I) = 0 GOTO 548
545 W = I : GOTO 535
548 U = I + 1 : GOTO 535
550 IF U >= D GOTO 515 : REM ONLY NULS, NO TEXT
555 C = C + 1 : L$(0) = STR$ (C) + ">"
560 FOR J = U TO D - 1 : X = PEEK (RX + J) : IF X < 32 GOTO 575
565 IF X > 126 THEN LET X = 46 : REM NON-ASCII -> "."
570 L$(0) = L$(0) + CHR$ (X) : IF LEN (L$(0)) < 26 GOTO 575
572 WAIT 96 : PRINT L$(0) : L$(0) = "" : REM FULL LINE: SHOW 1.5 S
575 NEXT J
580 WAIT 0 : IF LEN (L$(0)) > 0 PRINT L$(0)
585 BEEP 1 : GOTO 515
800 REM --- TX BUFFER STARTS WITH PL NUL BYTES, SO THE RECEIVER CAN SYNC ---
805 WAIT 0 : PRINT "TX "; BD; " "; P$; " ..."
810 FOR I = 0 TO PL - 1 : POKE TX + I, 0 : NEXT I : RETURN
850 REM --- T = SECONDS OF THE HOUR FROM TIME (MMDDHH.MMSS) ---
855 X = TIME : X = (X - INT (X)) * 100 : REM X = MM.SS
860 T = INT (X) * 60 + INT ((X - INT (X)) * 100 + .5) : RETURN
900 REM --- BACK TO THE MENU WHEN THE KEY IS RELEASED ---
905 WAIT 0 : PRINT "MENU"
910 G$ = INKEY$ : IF G$ <> "" GOTO 910
915 GOTO 200
950 POKE SI + 32, 18 : WAIT 0 : PRINT "BYE" : END
960 WAIT : PRINT "NO SERIN/SEROUT v6 CODE!"
965 PRINT "RUN INSTALLER v6.2 FIRST" : END
